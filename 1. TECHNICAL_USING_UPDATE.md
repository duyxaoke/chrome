```typescript
import { computed } from "@angular/core";
import { patchState, signalStore, withComputed, withMethods, withState } from "@ngrx/signals";

export interface BaseStoreState<T = any> {
  loading?: Record<string, boolean>;
  error?: Record<string, string | null>;
  data?: Record<string, T>;
  custom?: Record<string, any>;
}

export function createBaseStore<T>(
  initialState: BaseStoreState<T>,
  storeName: string
) {
  return signalStore(
    { providedIn: "root" },
    withState<BaseStoreState<T>>(initialState),
    
    withComputed(store => ({
      getData: computed(() => (key: string) => store.data()?.[key] ?? null),
      getLoading: computed(() => (key: string) => !!store.loading()?.[key]),
      getError: computed(() => (key: string) => store.error()?.[key] ?? null),
    })),
    
    withMethods(store => ({
      clearKey(key: string) {
        const { [key]: _l, ...restLoading } = store.loading() ?? {};
        const { [key]: _e, ...restError } = store.error() ?? {};
        const { [key]: _d, ...restData } = store.data() ?? {};

        patchState(store, {
          loading: restLoading,
          error: restError,
          data: restData
        });
      },

      clearAll() {
        patchState(store, { ...initialState });
      }
    }))
  );
}
```
