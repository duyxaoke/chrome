@Injectable({ providedIn: 'root' })
export class FilterBusinessService {
  
  constructor(
    private filterDataStore: FilterDataStore,
    private productSearchService: ProductSearchService,
    private productSearchTextStore: ProductSearchTextStore,
    private analyticsService: AnalyticsService
  ) {}

  handleFilterToggle(field_name: string, value: string | number, requestFilterBy: string) {
    // ✅ CENTRALIZED: All filter business logic here
    const updatedFilters = this.filterDataStore.toggleFilterSelection(field_name, value);
    
    const key = requestFilterBy 
      ? NEW_PRODUCT_SEARCH_TEXT_KEY_STORE.PRODUCT
      : NEW_PRODUCT_SEARCH_TEXT_KEY_STORE.DEFAULT;
    
    const facetCounts = this.productSearchTextStore.getFacetCounts()(key);
    
    if (facetCounts && Array.isArray(facetCounts)) {
      this.filterDataStore.updateVisibleFilters(facetCounts, updatedFilters);
    }

    this.navigateWithFilter(field_name, updatedFilters);
    this.trackFilterEvent(field_name, value);
  }

  private navigateWithFilter(field_name: string, updatedFilters: any[]) {
    const valueByFieldName = updatedFilters.find(item => item?.field_name === field_name)?.value;
    this.productSearchService.onChangeNavigate({
      key: field_name,
      value: valueByFieldName && valueByFieldName.length > 0 ? valueByFieldName : null
    });
  }

  private trackFilterEvent(field_name: string, value: string | number) {
    this.analyticsService.trackFilterSelection(field_name, value);
  }
}

// 🆕 CREATE: Centralized Error Handler

// File: services/error-handler.service.ts
@Injectable({ providedIn: 'root' })
export class ErrorHandlerService {
  
  constructor(private notificationService: NotificationService) {}
  
  handleStoreError(error: any, context: string): string {
    // ✅ CENTRALIZED: Log all errors
    console.error(`[${context}] Error:`, error);
    
    // ✅ CONSISTENT: Standard error categorization
    if (error?.status === 404) {
      return this.getErrorMessage('NOT_FOUND', context);
    }
    
    if (error?.status >= 500) {
      return this.getErrorMessage('SERVER_ERROR', context);
    }
    
    if (error?.status === 400) {
      return this.getErrorMessage('BAD_REQUEST', context);
    }
    
    if (error?.name === 'TimeoutError') {
      return this.getErrorMessage('TIMEOUT', context);
    }
    
    // Default error
    return error?.message || this.getErrorMessage('UNKNOWN', context);
  }
  
  private getErrorMessage(errorType: string, context: string): string {
    const errorMessages = {
      'NOT_FOUND': {
        'ProductSearchText': 'Không tìm thấy sản phẩm nào',
        'SearchImage': 'Không tìm thấy sản phẩm với hình ảnh này',
        'SuggestionSearch': 'Không có gợi ý tìm kiếm',
        'default': 'Không tìm thấy dữ liệu'
      },
      'SERVER_ERROR': {
        'default': 'Lỗi server, vui lòng thử lại sau'
      },
      'BAD_REQUEST': {
        'default': 'Yêu cầu không hợp lệ'
      },
      'TIMEOUT': {
        'default': 'Kết nối bị timeout, vui lòng thử lại'
      },
      'UNKNOWN': {
        'ProductSearchText': 'Có lỗi xảy ra khi tìm kiếm sản phẩm',
        'SearchImage': 'Có lỗi xảy ra khi tìm kiếm bằng hình ảnh',
        'SuggestionSearch': 'Có lỗi xảy ra khi tải gợi ý',
        'default': 'Có lỗi xảy ra, vui lòng thử lại'
      }
    };
    
    return errorMessages[errorType]?.[context] || errorMessages[errorType]?.['default'] || 'Có lỗi xảy ra';
  }
  
  createErrorHandler(store: any, key: string, context: string) {
    return catchError((err: any) => {
      const errorMessage = this.handleStoreError(err, context);
      
      // ✅ CONSISTENT: Same error handling pattern for all stores
      patchState(store, {
        loading: { ...store.loading(), [key]: false },
        error: { ...store.error(), [key]: errorMessage }
      });
      
      // ✅ OPTIONAL: Show user notification for critical errors
      if (err?.status >= 500) {
        this.notificationService.showError(errorMessage);
      }
      
      return of(null);
    });
  }
}
